# 3D Ground (stage 1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the world rock that can overhang, arch and hollow into caves, generate three new kinds of country with it (Stack Coast, Tablelands, Canyon Lands), and let the player walk over and under it.

**Architecture:** A tile keeps its ground height `h` and gains `slabs`, a list of `[lo, hi]` bands of rock above open space. A new pure module `landforms.js` plans stacks, islands, plateaus and mesas on per-layer grids from the seed and reshapes tiles through `Worldgen.natural`. Movement treats slab tops as surfaces and slab bodies as walls; the ground mesher draws every exposed face; the terrain shader bands the rock and thins it around the player.

**Tech Stack:** JavaScript (ES modules), three.js 0.187, Vite, Vitest. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-11-28-3d-ground-design.md`

## Global Constraints

- Every height is a multiple of `0.5`.
- `HEADROOM = 2`: the clear height under every slab. `PLAYER_HEIGHT = 1.5`.
- `tile.h` is always the lowest floor. Nothing that reads `h` today changes meaning.
- Slabs are never edited or saved in this stage. The save format does not change.
- No new country within 200 tiles of the world's centre, in the Frozen Wastes or Volcanic Lands, or on mountain tiles.
- Player-facing messages, verbatim: `Nothing can be built up here yet` and `No room under the rock`.
- Place names, verbatim: `Stack Coast`, `Tablelands`, `Canyon Lands`.
- No new dependencies. Comments say what the code does in plain words, in the style of the files around them.
- Run one test file with `npx vitest run tests/ground.test.js`; run everything with `npm test`.

```js
// One tile read as a column: solid ground at the bottom, and sometimes slabs of rock above it with
// open space between (an arch, a cave roof, an overhang). tile.slabs is [[lo, hi], ...], lowest first.
export const HEADROOM = 2; // clear height kept under every slab, in tiles
export const PLAYER_HEIGHT = 1.5; // how tall the player's body is

// The solid floor of a tile: the bed under water, the ground elsewhere.
export const floorOf = (t) => (t.water ? t.bed : t.h);

// Heights that can be stood on, lowest first.
export function surfaces(t) {
  const out = [floorOf(t)];
  if (t.slabs) for (const s of t.slabs) out.push(s[1]);
  return out;
}

// The top of the highest slab at or below `y`, or -Infinity when there is none.
export function slabTopBelow(t, y) {
  let top = -Infinity;
  if (t.slabs) for (const s of t.slabs) if (s[1] <= y) top = s[1];
  return top;
}

// The underside of the first slab that reaches above `y`, or Infinity under open sky.
export function ceilingAbove(t, y) {
  if (t.slabs) for (const s of t.slabs) if (s[1] > y) return s[0];
  return Infinity;
}

// Does any slab lie between two heights? (The ground itself is the caller's business.)
export function solidAt(t, y0, y1) {
  if (t.slabs) for (const s of t.slabs) if (s[0] < y1 && s[1] > y0) return true;
  return false;
}

// Every solid range of the tile, lowest first. The ground reaches down for ever.
export function solids(t) {
  return t.slabs ? [[-Infinity, floorOf(t)], ...t.slabs] : [[-Infinity, floorOf(t)]];
}
```

```js
// Worlds with nothing in them but what the landforms need: all land, or all shallow sea.
const allLand = (seed) => new Landforms({ seed, sea: 0, rugged: 1, continent: () => 1, plain: () => true });
const allSea = (seed) => new Landforms({ seed, sea: 0, rugged: 1, continent: () => -0.01, plain: () => true });
const GROUND = { h: 0, water: false, salt: false, bed: null };
const SEA = { h: -0.3, water: true, salt: true, bed: -4 };
const topOf = (form) => Math.max(form.h ?? -Infinity, ...(form.slabs ?? []).map((s) => s[1]));
const of = (list, id) => list.filter((f) => f.layer.id === id);

// Do the slabs of a reshaped tile obey the rules? floor: the ground they stand over.
function slabsAreSound(slabs, floor) {
  for (const [lo, hi] of slabs) {
    if ((lo * 2) % 1 || (hi * 2) % 1 || hi - lo < 0.5 || lo < floor + HEADROOM) return false;
    floor = hi;
  }
  return true;
}
```

```js
import { fbm, hash2 } from '../core/noise.js';
import { HEADROOM } from './columns.js';

const TAU = Math.PI * 2;
const half = (v) => Math.round(v * 2) / 2;
const floorHalf = (v) => Math.floor(v * 2) / 2;
const RAMP = 0.4; // height a route gains for each tile of its length: always less than one half-tile step
const TRAIL = 3; // width of a route, in tiles
const LANDING = 5; // level tiles at the top of a route
const RIM = 10; // how far in from a route the rock is left whole
const SKIRT = 6; // tiles around a landform that still count as its country

// One possible landform per square of `cell` tiles, so two of a kind never overlap.
//   r, h       smallest and largest radius and height
//   amp, grain how ragged the outline is, and at what scale
//   lip        widest overhang at the top of the cliff, in tiles
//   apron      beach and shallows laid around the foot, in tiles
// Listed largest first: a tile takes the name of the largest landform that reaches it.
export const LAYERS = [
  { id: 'mesa', land: 'canyon', cell: 500, chance: 0.85, r: [110, 200], h: [30, 50], amp: 0.12, grain: 60, lip: 3, apron: 0, ramp: true, salt: 400 },
  { id: 'plateau', land: 'table', cell: 380, chance: 0.8, r: [40, 150], h: [8, 25], amp: 0.2, grain: 26, lip: 2, apron: 0, ramp: true, tiers: true, salt: 420 },
  { id: 'island', land: 'stacks', cell: 170, chance: 0.65, r: [20, 30], h: [14, 35], amp: 0.14, grain: 16, lip: 3, apron: 12, ramp: true, salt: 440 },
  { id: 'stack', land: 'stacks', cell: 44, chance: 0.6, r: [3, 11], h: [12, 35], amp: 0.22, grain: 6, lip: 3, apron: 0, ramp: false, salt: 460 },
];
const STACK = LAYERS[3];
const PILLAR_CELL = 36, BRIDGE_CELL = 90;

// Tidy the rock above one tile so it obeys the rules: sorted, never touching, and with HEADROOM
// under every slab. Rock too close to the ground joins it. -> { h, slabs } (slabs undefined if none)
export function settle(h, slabs, water) {
  const sorted = slabs.filter(([lo, hi]) => hi - lo >= 0.5).sort((a, b) => a[0] - b[0]);
  const out = [];
  for (const [lo, hi] of sorted) {
    const last = out[out.length - 1];
    if (last && lo - last[1] < HEADROOM) last[1] = Math.max(last[1], hi);
    else out.push([lo, hi]);
  }
  while (out.length) {
    const [lo, hi] = out[0];
    if (water) {
      // over water the gap is measured from the surface; rock never fills the sea in
      if (lo >= HEADROOM) break;
      if (hi - HEADROOM >= 0.5) { out[0][0] = HEADROOM; break; }
      out.shift();
    } else {
      if (lo >= h + HEADROOM) break;
      if (hi > h) h = hi;
      out.shift();
    }
  }
  return { h, slabs: out.length ? out : undefined };
}

// The stacks, islands, plateaus and canyon mesas of a world, worked out from the seed alone.
// world: { seed, sea, rugged, continent(x, z), plain(x, z) }
//   sea       land noise below this is sea (from the world type)
//   rugged    how common the new country is, 1 for the usual
//   plain     true where nothing else has shaped the land: no mountain, no frost, no heat
export class Landforms {
  constructor(world) {
    this.world = world;
    this.cache = new Map();
  }

  remember(key, make) {
    let v = this.cache.get(key);
    if (v === undefined) {
      if (this.cache.size > 20000) this.cache.clear();
      this.cache.set(key, (v = make()));
    }
    return v;
  }

  // The landform of one layer in one square of its grid, or null.
  feature(layer, cx, cz) {
    return this.remember(layer.id + ':' + cx + ',' + cz, () => this.plan(layer, cx, cz));
  }

  plan(layer, cx, cz) {
    const s = this.world.seed + layer.salt;
    const roll = (n) => hash2(cx, cz, s + n);
    if (roll(0) >= layer.chance) return null;
    const r = layer.r[0] + (layer.r[1] - layer.r[0]) * (layer.id === 'stack' ? roll(1) ** 2 : roll(1));
    const reach = r * (1 + layer.amp) + layer.lip + 1 + layer.apron;
    const room = layer.cell / 2 - reach;
    const x = Math.round((cx + 0.5) * layer.cell + room * (roll(2) * 2 - 1));
    const z = Math.round((cz + 0.5) * layer.cell + room * (roll(3) * 2 - 1));
    if (Math.hypot(x, z) - reach - SKIRT < 200) return null; // the heart of the world stays gentle
    if (!this.suits(layer, x, z, r)) return null;
    // never touching a larger landform: each keeps its own cliffs and its own way up
    for (const big of LAYERS.slice(0, LAYERS.indexOf(layer))) {
      const span = reach + big.cell / 2;
      for (let bz = Math.floor((z - span) / big.cell); bz <= Math.floor((z + span) / big.cell); bz++) {
        for (let bx = Math.floor((x - span) / big.cell); bx <= Math.floor((x + span) / big.cell); bx++) {
          const b = this.feature(big, bx, bz);
          if (b && Math.hypot(b.x - x, b.z - z) < b.reach + reach + SKIRT) return null;
        }
      }
    }
    let h = half(layer.h[0] + (layer.h[1] - layer.h[0]) * roll(4));
    // a route may wind at most four fifths of the way round: small islands cannot be very tall
    if (layer.ramp) h = Math.min(h, floorHalf(0.8 * TAU * RAMP * (r * (1 - layer.amp) - TRAIL - 1)));
    const count = layer.tiers ? Math.max(1, Math.min(r >= 60 ? 3 : 2, Math.floor(h / 4), 1 + Math.floor(roll(5) * 3))) : 1;
    const tiers = [];
    let a0 = roll(6) * TAU;
    for (let i = 0; i < count; i++) {
      const tr = r * 0.62 ** i, rho = tr * (1 - layer.amp) - TRAIL - 1;
      const lo = half((h * i) / count), hi = half((h * (i + 1)) / count);
      const span = (hi - lo) / RAMP / rho;
      tiers.push({ r: tr, lo, hi, rho, a0, span });
      a0 += span + (TAU - span) / 2; // the way up the next tier starts on the far side
    }
    return { layer, x, z, r, h, reach, tiers, cx, cz };
  }

  // Is this the right kind of place for a landform of this layer?
  suits(layer, x, z, r) {
    const { seed: s, sea, rugged, continent, plain } = this.world;
    const ease = 0.12 * (rugged - 1);
    if (layer.land === 'stacks') {
      // off the coast or right on it, along chosen stretches of shore
      if (fbm(x / 800, z / 800, s + 72, 2) < 0.02 - ease) return false;
      const c = continent(x, z);
      return c > sea - 0.03 && c < sea + 0.003 && plain(x, z);
    }
    // canyon country and tableland each have their own stretches; where both would be, it is canyon
    const canyon = fbm(x / 1300, z / 1300, s + 71, 2) > 0.04 - ease;
    if (layer.id === 'mesa' ? !canyon : canyon || fbm(x / 1300, z / 1300, s + 70, 2) < 0.04 - ease) return false;
    // wholly on land that nothing else has shaped
    const edge = r * (1 + layer.amp);
    for (let k = 0; k < 16; k++) {
      const a = (k / 8) * Math.PI * 2, d = k < 8 ? edge : edge * 0.5;
      const px = x + Math.cos(a) * d, pz = z + Math.sin(a) * d;
      if (continent(px, pz) <= sea + 0.004 || !plain(px, pz)) return false;
    }
    return continent(x, z) > sea + 0.004 && plain(x, z);
  }

  // The canyon cut through a mesa at this spot: 0 on the canyon floor, 1 to 3 on the terraces of
  // its walls, 4 on the uncut top.
  canyon(x, z) {
    const s = this.world.seed;
    const a = Math.abs(fbm(x / 130, z / 130, s + 81, 2));
    const w = 0.03 + 0.03 * (0.5 + 0.5 * fbm(x / 400, z / 400, s + 82, 1));
    return a < w ? 0 : Math.min(4, 1 + Math.floor((a - w) / 0.02));
  }

  // A rock pillar standing on a canyon floor, or null: one possible per small square.
  pillar(f, cx, cz) {
    return this.remember('pillar:' + cx + ',' + cz, () => {
      const s = this.world.seed + 480;
      if (hash2(cx, cz, s) > 0.5) return null;
      const x = cx * PILLAR_CELL + 8 + Math.floor(hash2(cx, cz, s + 1) * (PILLAR_CELL - 16));
      const z = cz * PILLAR_CELL + 8 + Math.floor(hash2(cx, cz, s + 2) * (PILLAR_CELL - 16));
      if (this.canyon(x, z) !== 0) return null;
      return { x, z, neck: 2 + hash2(cx, cz, s + 3) * 1.5, h: half(f.h * (0.45 + 0.3 * hash2(cx, cz, s + 4))) };
    });
  }

  // A natural bridge across a canyon, or null: one possible per square.
  bridge(f, cx, cz) {
    return this.remember('bridge:' + cx + ',' + cz, () => {
      const s = this.world.seed + 490;
      if (hash2(cx, cz, s) > 0.9) return null;
      const x = cx * BRIDGE_CELL + 30 + Math.floor(hash2(cx, cz, s + 1) * (BRIDGE_CELL - 60));
      const z = cz * BRIDGE_CELL + 30 + Math.floor(hash2(cx, cz, s + 2) * (BRIDGE_CELL - 60));
      if (this.canyon(x, z) !== 0) return null;
      // it runs straight across: along the steepest way out of the canyon
      const n = (px, pz) => Math.abs(fbm(px / 130, pz / 130, this.world.seed + 81, 2));
      const gx = n(x + 3, z) - n(x - 3, z), gz = n(x, z + 3) - n(x, z - 3), g = Math.hypot(gx, gz);
      if (g < 1e-4) return null;
      return { x, z, dx: gx / g, dz: gz / g, len: 28, top: half(f.h / 2) };
    });
  }

  // A sea arch from the stack in one square to the stack in the next (axis 0: +x, 1: +z), or null.
  arch(cx, cz, axis) {
    return this.remember('arch:' + cx + ',' + cz + ',' + axis, () => {
      const a = this.feature(STACK, cx, cz), b = this.feature(STACK, cx + (axis ? 0 : 1), cz + (axis ? 1 : 0));
      if (!a || !b || hash2(cx, cz, this.world.seed + 470 + axis) > 0.7) return null;
      const span = Math.hypot(b.x - a.x, b.z - a.z), gap = span - a.r - b.r;
      if (gap < 5 || gap > 30) return null;
      return { x: a.x, z: a.z, dx: (b.x - a.x) / span, dz: (b.z - a.z) / span, from: a.r * 0.5, to: span - b.r * 0.5, top: Math.min(a.h, b.h) };
    });
  }

  // Every landform whose middle is within `radius` tiles of a spot, nearest first.
  near(x, z, radius) {
    const out = [];
    for (const layer of LAYERS) {
      for (let cz = Math.floor((z - radius) / layer.cell); cz <= Math.floor((z + radius) / layer.cell); cz++) {
        for (let cx = Math.floor((x - radius) / layer.cell); cx <= Math.floor((x + radius) / layer.cell); cx++) {
          const f = this.feature(layer, cx, cz);
          if (f && Math.hypot(f.x - x, f.z - z) <= radius) out.push(f);
        }
      }
    }
    return out.sort((a, b) => Math.hypot(a.x - x, a.z - z) - Math.hypot(b.x - x, b.z - z));
  }

  // How the landforms reshape one tile, or null where there are none.
  // base: the tile before them: { h, water, salt, bed }
  // -> { land, h, slabs, dry, bed, shore, biome, path }; only the fields that change are set
  at(x, z, base) {
    let land = null, rock = 0, path = null, beach = false, shallow = false, biome = null;
    const slabs = [];
    for (const layer of LAYERS) {
      const f = this.feature(layer, Math.floor(x / layer.cell), Math.floor(z / layer.cell));
      if (!f) continue;
      const dx = x - f.x, dz = z - f.z, dist = Math.hypot(dx, dz);
      if (dist > f.reach + SKIRT) continue;
      land ??= layer.land;
      const s = this.world.seed + layer.salt;
      const rag = Math.max(-1, Math.min(1, fbm(x / layer.grain, z / layer.grain, s + 10, 3) * 1.6));
      const scale = 1 + layer.amp * rag;
      // the routes up: a level cut in from outside, then a trail winding up just inside the edge
      let whole = false;
      if (layer.ramp) {
        for (const t of f.tiers) {
          const c = Math.cos(t.a0), sn = Math.sin(t.a0);
          const along = dx * c + dz * sn, across = Math.abs(dz * c - dx * sn);
          if (across <= TRAIL / 2 && along >= t.rho && along <= t.r * (1 + layer.amp) + layer.lip + 1) path = Math.max(path ?? 0, t.lo);
          if (dist < t.rho - RIM || dist >= t.r * (1 + layer.amp)) continue;
          const rel = (((Math.atan2(dz, dx) - t.a0) % TAU) + TAU) % TAU;
          // the trail ends on a short landing at the top, and the rim beside it is never cut by a canyon
          if (dist >= t.rho && dist < t.rho + TRAIL && rel <= t.span + LANDING / t.rho) {
            path = Math.max(path ?? 0, t.lo + Math.min(t.hi - t.lo, floorHalf(RAMP * rel * t.rho + 1e-6)));
          }
          if (rel <= t.span + 0.3 || rel >= TAU - 0.1) whole = true;
        }
      }
      let top = 0;
      for (let i = f.tiers.length - 1; i >= 0 && !top; i--) if (dist < f.tiers[i].r * scale) top = f.tiers[i].hi;
      const out = dist - f.r * scale; // tiles beyond the foot of the cliff (negative inside)
      if (layer.id === 'mesa' && out < SKIRT) biome = 'canyon';
      if (!top) {
        // the top juts out past the foot in places, thinning toward its lip
        const n = fbm(x / 9, z / 9, s + 15, 2);
        const wide = n > 0 ? Math.min(layer.lip, 1 + Math.floor(n * 8)) : 0;
        if (out < wide) slabs.push([f.tiers[0].hi - Math.max(1, 2 + wide - 1.5 * Math.floor(out)), f.tiers[0].hi]);
        if (layer.apron && out < layer.apron / 2) beach = true;
        else if (layer.apron && out < layer.apron) shallow = true;
        continue;
      }
      if (layer.id === 'stack' && base.salt && out > -1.2 && top >= 5) {
        // the sea has eaten a notch into the foot of the stack
        slabs.push([HEADROOM, top]);
        continue;
      }
      if (layer.id === 'mesa') {
        if (!whole) top = Math.min(top, half((f.h * this.canyon(x, z)) / 4));
        const p = this.pillar(f, Math.floor(x / PILLAR_CELL), Math.floor(z / PILLAR_CELL));
        if (p) {
          const d = Math.hypot(x - p.x, z - p.z);
          if (d < p.neck) top = Math.max(top, p.h);
          else if (d < p.neck + 2.2) slabs.push([p.h - 3 + Math.floor(d - p.neck), p.h]);
        }
        const b = this.bridge(f, Math.floor(x / BRIDGE_CELL), Math.floor(z / BRIDGE_CELL));
        if (b) {
          const along = (x - b.x) * b.dx + (z - b.z) * b.dz, across = Math.abs((z - b.z) * b.dx - (x - b.x) * b.dz);
          if (Math.abs(along) <= b.len && across <= TRAIL / 2) slabs.push([b.top - 2 - half(2 * (along / b.len) ** 2), b.top]);
        }
      }
      rock = Math.max(rock, top);
    }
    // sea arches between neighbouring stacks
    const cx = Math.floor(x / STACK.cell), cz = Math.floor(z / STACK.cell);
    for (const [ax, az, axis] of [[cx, cz, 0], [cx, cz, 1], [cx - 1, cz, 0], [cx, cz - 1, 1]]) {
      const a = this.arch(ax, az, axis);
      if (!a) continue;
      const along = (x - a.x) * a.dx + (z - a.z) * a.dz, across = Math.abs((z - a.z) * a.dx - (x - a.x) * a.dz);
      if (along < a.from || along > a.to || across > TRAIL / 2) continue;
      const u = (along - (a.from + a.to) / 2) / ((a.to - a.from) / 2);
      slabs.push([half(a.top - 2 - (a.top - 2 - HEADROOM) * u * u), a.top]);
      land ??= 'stacks';
    }
    if (!land) return null;
    const ground = base.water ? 0 : base.h;
    // a route is kept clear of everything: no rock over it, nothing growing on it
    if (path !== null) return path > 0 ? { land, path: true, dry: true, h: Math.max(ground, path), biome } : { land, path: true, biome };
    if (rock > 0) {
      // caves: winding passages at ground level through any rock tall enough to hold them
      const s = this.world.seed;
      const vein = Math.abs(fbm(x / 45, z / 45, s + 85, 2));
      if (vein < 0.03 && fbm(x / 260, z / 260, s + 86, 2) > 0) {
        const roof = ground + 3 + half(2 * (1 - vein / 0.03));
        if (rock >= roof + 2) return { land, biome, ...settle(base.h, [...slabs, [roof, rock]], base.water) };
      }
      return { land, biome, dry: true, ...settle(rock, slabs, false) };
    }
    if (base.water && beach) return { land, biome, dry: true, shore: true, ...settle(0, slabs, false) };
    const out = { land, biome, ...settle(base.h, slabs, base.water) };
    if (base.water && shallow) out.bed = Math.max(base.bed, -0.9);
    return out;
  }
}
```

```js
import { hash2 } from '../core/noise.js';
import { BIOMES, SHORE, WATER_COLORS, SEABED_COLORS, LAVA_COLORS, ICE_COLORS } from '../world/biomes.js';
import { WATER_Y } from '../world/worldgen.js';
import { floorOf, solids } from '../world/columns.js';
import { rgb } from './voxel.js';

export const CHUNK = 16;

const linearPair = (pair) => pair.map(rgb);
const GROUND = Object.fromEntries(Object.entries(BIOMES).map(([k, b]) => [k, linearPair(b.ground)]));
const SHORE_GROUND = linearPair(SHORE.ground);
const WATER = linearPair(WATER_COLORS);
const SIDE_TINT = rgb(0x9a7a5a);
const DIRT = linearPair([0x8f633f, 0x7a5233]); // ground the player has dug shows bare earth
const SEABED = [rgb(SEABED_COLORS.shallow), rgb(SEABED_COLORS.deep)];
const LAVA = linearPair(LAVA_COLORS), ICE = linearPair(ICE_COLORS), BOIL = linearPair([0xd9703a, 0xc25a2e]);
const STONE = linearPair([0xb8b4a8, 0xa39f94]); // bare patches on the tops of the stacks
const UNDERSIDE = rgb(0x3b3632); // the underside of rock
const ROCK = { stacks: 1, table: 2, canyon: 3 }; // which layered rock a cliff is drawn in (see the terrain shader)
const DEEP = 0.62; // a bank is darker at its foot than at its lip
const WHITE = [1, 1, 1];
const AROUND = [[1, 0], [-1, 0], [0, 1], [0, -1]];

// The parts of the range a..b that none of `covers` (ranges, lowest first) fill.
export function gaps(a, b, covers) {
  const out = [];
  for (const [lo, hi] of covers) {
    if (hi <= a) continue;
    if (lo >= b) break;
    if (lo > a) out.push([a, lo]);
    a = Math.max(a, hi);
    if (a >= b) break;
  }
  if (a < b) out.push([a, b]);
  return out;
}

// How deep under rock a tile is: 0 in the open, up to 4 far inside a cave.
function cover(gen, x, z) {
  if (!gen.tile(x, z).slabs) return 0;
  let best = 4;
  for (const [dx, dz] of AROUND) {
    for (let n = 1; n < best; n++) {
      if (!gen.tile(x + dx * n, z + dz * n).slabs) {
        best = n;
        break;
      }
    }
  }
  return best;
}
const dimmed = (depth) => 1 - 0.13 * depth;

function maker(withRock = false) {
  const pos = [], norm = [], col = [], idx = [], rock = [];
  // light: how bright each of the four corners is (1 = full), for shading that fades across the square
  // kind: 0 for an ordinary face, or which layered rock the shader should draw it in
  const quad = (corners, n, c, light = null, kind = 0) => {
    const base = pos.length / 3;
    corners.forEach((p, i) => {
      const k = light ? light[i] : 1;
      pos.push(p[0], p[1], p[2]);
      norm.push(n[0], n[1], n[2]);
      col.push(c[0] * k, c[1] * k, c[2] * k);
      if (withRock) rock.push(kind);
    });
    if (!light || light[0] + light[2] >= light[1] + light[3]) idx.push(base, base + 1, base + 2, base, base + 2, base + 3);
    else idx.push(base + 1, base + 2, base + 3, base + 1, base + 3, base);
  };
  const done = () => {
    const out = { pos: new Float32Array(pos), norm: new Float32Array(norm), col: new Float32Array(col), idx: new Uint32Array(idx) };
    if (withRock) out.rock = new Float32Array(rock);
    return out;
  };
  return { quad, done };
}

// One upright face of a tile toward a neighbour (dir: 0 +x, 1 -x, 2 +z, 3 -z), from y0 up to y1,
// wound to face outward. lo, hi: how bright its foot and its lip are.
function wall(land, dir, x, z, y0, y1, c, lo, hi, kind) {
  if (dir === 0) land.quad([[x + 1, y0, z], [x + 1, y1, z], [x + 1, y1, z + 1], [x + 1, y0, z + 1]], [1, 0, 0], c, [lo, hi, hi, lo], kind);
  else if (dir === 1) land.quad([[x, y0, z], [x, y0, z + 1], [x, y1, z + 1], [x, y1, z]], [-1, 0, 0], c, [lo, lo, hi, hi], kind);
  else if (dir === 2) land.quad([[x, y0, z + 1], [x + 1, y0, z + 1], [x + 1, y1, z + 1], [x, y1, z + 1]], [0, 0, 1], c, [lo, lo, hi, hi], kind);
  else land.quad([[x, y0, z], [x, y1, z], [x + 1, y1, z], [x + 1, y0, z]], [0, 0, -1], c, [lo, hi, hi, lo], kind);
}

// The ground of one chunk as plain arrays, ready for toGeometry.
// -> { land, water, lava }: the solid ground (with sea and pond beds and all rock), the see-through
// water surface, and the glowing lava.
export function buildTerrain(gen, cx, cz) {
  const land = maker(true), water = maker(), lava = maker();
  for (let z = cz * CHUNK; z < (cz + 1) * CHUNK; z++) {
    for (let x = cx * CHUNK; x < (cx + 1) * CHUNK; x++) {
      const t = gen.tile(x, z);
      const k = hash2(x, z, 31), shade = 0.95 + hash2(x, z, 32) * 0.1;
      // the colour of open ground here: also the colour of the top of any rock above it
      const sky = t.land === 'stacks' && t.h >= 6 && hash2(x >> 1, z >> 1, 77) > 0.72 ? STONE : GROUND[t.biome === 'ocean' ? 'meadow' : t.biome];
      const top = [0, 1, 2].map((i) => (sky[0][i] + (sky[1][i] - sky[0][i]) * k) * shade);
      let c;
      if (t.water) {
        // bed: sandy in the shallows, fading to dark blue-grey in the deep
        const depth = Math.min(1, (WATER_Y - t.bed) / 9);
        c = [0, 1, 2].map((i) => (SEABED[0][i] + (SEABED[1][i] - SEABED[0][i]) * depth) * shade);
        // the boiling sea of the volcanic region is a murky orange-red
        const w = t.boiling
          ? [0, 1, 2].map((i) => BOIL[0][i] + (BOIL[1][i] - BOIL[0][i]) * k)
          : [0, 1, 2].map((i) => (WATER[0][i] + (WATER[1][i] - WATER[0][i]) * k) * (1 - depth * 0.35));
        // paler where it laps against land: a line of foam along every shore
        const foam = (gx, gz) => {
          for (const [ox, oz] of [[-1, -1], [0, -1], [-1, 0], [0, 0]]) if (!gen.tile(gx + ox, gz + oz).water) return 1.55;
          return 1;
        };
        water.quad([[x, WATER_Y, z], [x, WATER_Y, z + 1], [x + 1, WATER_Y, z + 1], [x + 1, WATER_Y, z]], [0, 1, 0], w, [foam(x, z), foam(x, z + 1), foam(x + 1, z + 1), foam(x + 1, z)]);
      } else {
        const pair = t.dug < 0 ? DIRT : t.ice ? ICE : t.shore ? SHORE_GROUND : t.slabs ? GROUND[t.biome] : sky;
        c = [0, 1, 2].map((i) => (pair[0][i] + (pair[1][i] - pair[0][i]) * k) * shade);
        // lava glows: it gets its own unlit layer just above the rock
        if (t.lava) {
          const g = [0, 1, 2].map((i) => LAVA[0][i] + (LAVA[1][i] - LAVA[0][i]) * k);
          lava.quad([[x, t.h + 0.03, z], [x, t.h + 0.03, z + 1], [x + 1, t.h + 0.03, z + 1], [x + 1, t.h + 0.03, z]], [0, 1, 0], g);
        }
      }
      const h = floorOf(t);
      // under rock the ground is dim, and dimmer the farther it is from open sky
      const dim = t.slabs ? dimmed(cover(gen, x, z)) : 1;
      // the ground darkens a little at the foot of anything taller beside it: a bank, a cliff, a shore
      const nook = (gx, gz) => {
        let n = 0;
        for (const [ox, oz] of [[-1, -1], [0, -1], [-1, 0], [0, 0]]) if (floorOf(gen.tile(gx + ox, gz + oz)) > h + 0.2) n++;
        return (1 - n * 0.13) * dim;
      };
      land.quad([[x, h, z], [x, h, z + 1], [x + 1, h, z + 1], [x + 1, h, z]], [0, 1, 0], c, [nook(x, z), nook(x, z + 1), nook(x + 1, z + 1), nook(x + 1, z)]);

      // rock above the ground: a top in the colour of open ground, and a dark underside
      if (t.slabs) {
        for (const [lo, hi] of t.slabs) {
          land.quad([[x, hi, z], [x, hi, z + 1], [x + 1, hi, z + 1], [x + 1, hi, z]], [0, 1, 0], top);
          land.quad([[x, lo, z], [x + 1, lo, z], [x + 1, lo, z + 1], [x, lo, z + 1]], [0, -1, 0], UNDERSIDE);
        }
      }

      // Upright faces toward each neighbour: every part of this tile's rock that the neighbour's
      // rock does not cover. In rugged country they are layered rock under a lip in the colour of
      // the top; elsewhere a bank in the darkened colour of the ground.
      const kind = ROCK[t.land] ?? 0;
      const mine = t.slabs ? solids(t) : null;
      AROUND.forEach(([dx, dz], dir) => {
        const nt = gen.tile(x + dx, z + dz), nh = floorOf(nt);
        if (!mine && !nt.slabs && nh >= h) return; // the usual case: nothing to draw
        const theirs = solids(nt);
        // a face seen from under the neighbour's rock is as dim as the ground there
        const roofed = nt.slabs ? dimmed(cover(gen, x + dx, z + dz)) : 1;
        (mine ?? [[-Infinity, h]]).forEach(([a, b], i) => {
          const lipColour = i === 0 ? c : top; // the ground's own colour, or the grass on top of a slab
          const bank = lipColour.map((v, j) => v * 0.45 + SIDE_TINT[j] * 0.3);
          for (const [y0, y1] of gaps(a, b, theirs)) {
            const under = nt.slabs && y1 <= nt.slabs[0][0] ? roofed : 1;
            if (!kind) {
              wall(land, dir, x, z, y0, y1, bank, DEEP * under, under, 0);
              continue;
            }
            // the top half tile of a cliff keeps the colour of what grows above it
            const lip = y1 === b ? Math.max(y0, y1 - 0.5) : y1;
            if (lip < y1) wall(land, dir, x, z, lip, y1, bank, under, under, 0);
            if (y0 < lip) wall(land, dir, x, z, y0, lip, WHITE, 0.8 * under, under, kind);
          }
        });
      });
    }
  }
  return { land: land.done(), water: water.done(), lava: lava.done() };
}
```
